#include<iostream>
using namespace std;
int n,m;
int st,ed;
int a[305][305] = {};
int dis[305][305] = {};
int v[305][305] = {};
int d1[4] = {0,0,1,-1};
int d2[4] = {1,-1,0,0};

// ---- min-heap thủ công (lazy deletion) ----
struct Node { int dist, i, j; };
Node heap[400005];
int heapSize = 0;

void heapPush(int dist, int i, int j)
{
	heap[++heapSize] = {dist, i, j};
	int c = heapSize;
	while (c > 1)
	{
		int p = c / 2;
		if (heap[p].dist <= heap[c].dist) break;
		Node tmp = heap[p]; heap[p] = heap[c]; heap[c] = tmp;
		c = p;
	}
}

Node heapPop()
{
	Node top = heap[1];
	heap[1] = heap[heapSize--];
	int p = 1;
	while (true)
	{
		int l = p*2, r = p*2+1, small = p;
		if (l <= heapSize && heap[l].dist < heap[small].dist) small = l;
		if (r <= heapSize && heap[r].dist < heap[small].dist) small = r;
		if (small == p) break;
		Node tmp = heap[p]; heap[p] = heap[small]; heap[small] = tmp;
		p = small;
	}
	return top;
}
// --------------------------------------------

void djk(int si,int sj)
{
	heapSize = 0;
	for (int i = 0; i < 4; i++)
	{
		int xi = si + d1[i];
		int xj = sj + d2[i];
		if (xi < 0 || xi >= m || xj < 0 || xj >=n) continue;
		if (a[xi][xj] != -1)
		{
			dis[xi][xj] = a[xi][xj];
			heapPush(dis[xi][xj], xi, xj);
		}
	}
	v[si][sj] = 1;

	while (heapSize > 0)
	{
		Node cur = heapPop();
		if (v[cur.i][cur.j]) continue;          // đã xử lý rồi -> bỏ qua (lazy delete)
		if (cur.dist > dis[cur.i][cur.j]) continue; // entry cũ, đã bị update -> bỏ qua
		v[cur.i][cur.j] = 1;
		if (cur.i == st && cur.j == ed) return;

		for (int i = 0; i < 4; i++)
		{
			int xi = cur.i + d1[i];
			int xj = cur.j + d2[i];
			if (xi < 0 || xi >= m || xj < 0 || xj >=n) continue;
			if (!v[xi][xj] && a[xi][xj] != -1 && dis[xi][xj] > dis[cur.i][cur.j] + a[xi][xj])
			{
				dis[xi][xj] = dis[cur.i][cur.j] + a[xi][xj];
				heapPush(dis[xi][xj], xi, xj);
			}
		}
	}
	return;
}

int main()
{
	int t;
	cin>>t;
	for(int test_case = 1; test_case <= t; ++test_case)
	{
		st = ed = 0;
		cin>>m>>n;
		for (int i = 0; i < m; i++)
		{
			for (int j = 0; j < n; j++)
			{
				char c;
				cin>>c;
				if (c == 'Y') a[i][j] = 0;
				else if (c == 'B') a[i][j] = 2;
				else if (c == 'E') a[i][j] = 1;
				else if (c == 'R' || c == 'S') a[i][j] = -1;
				else if (c == 'T')
				{
					a[i][j] = 1;
					st = i;
					ed = j;
				}
				v[i][j] = 0;
				dis[i][j] = 10000000;
			}
		}

		djk(0,0);
		if (dis[st][ed] == 10000000) cout<<"Case #"<<test_case<<endl<<"-1"<<endl;
		else
			cout<<"Case #"<<test_case<<endl<<dis[st][ed]<<endl;
	}
	return 0;
}
