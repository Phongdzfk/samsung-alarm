#include <iostream>
using namespace std;
int n;
int num[25];
int cost[25];
int fatig[25];
int choose[25];
int dx[8] = {0,0,1,-1,1,1,-1,-1};
int dy[8] = {1,-1,0,0,1,-1,1,-1};
int ans;
int maxcnt;
void quaylui(int cong,int total, int binhsi)
{
	if (cong == n+1)
	{
		if (total < ans) ans = total;
		return;
	}
	if (total > ans) return;
	//pass
	quaylui(cong+1,total + cost[cong],binhsi);
	//hire
	choose[cong] = num[cong];
	fatig[cong] = 3;
	quaylui(cong+1,total + cost[cong]*2,binhsi + num[cong]);
	choose[cong] = 0;
	fatig[cong] = 0;
	//battle
	int opp = num[cong];
	for (int i = 1; i <= n; i++)
	{
		if (choose[i] != 0 && fatig[i] == 0) 
		{
			choose[i] = 0;
			binhsi-=num[i];
		}
	}
	int tmp = binhsi;
	int tmp1 = opp;
	if (binhsi >= opp)
	{
		tmp -= opp;
		for (int i = 1; i <= n; i++)
		{
			if (choose[i] != 0)
			{
				fatig[i]-=1;
				if (opp > 0)
				{
					if (opp > choose[i])
					{
						choose[i] = 0;
						opp-=choose[i];
					}
					else
					{
						choose[i] = choose[i] - opp;
						opp = 0;
					}
				}
			}
			quaylui(cong+1,total,tmp);
			for (int i = 1; i <= n; i++)
			{
				if (choose[i] != 0)
				{
					fatig[i]+=1;
					if (tmp1 > 0)
					{
						if (tmp1 > choose[i])
						{
							choose[i] = num[i];
							tmp-=choose[i];
						}
						else
						{
							choose[i] = choose[i] + tmp;
							tmp = 0;
						}
					}
				}
			}
		}
	}
}

int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	//freopen("Text.txt","r",stdin);
	int t;
	cin>>t;
	for (int test=1; test<=t; test++)
	{
		cin>>n;
		ans = 1e9;
		for (int i = 1; i <=n; i++)
		{
			cin>>num[i]>>cost[i];
			choose[i] = 0;
		}
		quaylui(1,0,0);
		cout<<"#"<<test<<" "<<ans<<endl;
	}
	return 0;
}
