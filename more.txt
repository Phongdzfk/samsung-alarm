#include <iostream>
using namespace std;
char a[205][205];
int visited[205][205][2];   // [i][j][0]=dọc, [i][j][1]=ngang
int d1[4] = {0,0,-1,1};
int d2[4] = {1,-1,0,0};
int n,m;
struct point
{
	int a;
	int b;
	int dir;
};
point q[10000005];
int front = 0;
int rear = -1;
void push(int i,int j,int dir)
{
	rear++;
	q[rear].a = i;
	q[rear].b = j;
	q[rear].dir = dir;
}
point pop()
{
	return q[front++];
}
bool empty()
{
	return front > rear;
}
void bfs(int i1, int j1, int i2, int j2)
{
	visited[i1][j1][0] = 0;
	visited[i1][j1][1] = 0;
	push(i1,j1,0);
	push(i1,j1,1);   // ô xuất phát: chưa có hướng, coi cả 2 trạng thái đều = 0
	while (!empty())
	{
		point z = pop();
		for (int i = 0; i<4; i++)
		{
			int x1 = z.a + d1[i];
			int x2 = z.b + d2[i];
			if (x1 < 0 || x1 >= m || x2 < 0|| x2 >= n) continue;
			if (a[x1][x2] == '0')
			{
				int newDir = (d1[i] != 0) ? 0 : 1;
				int cost = (newDir == z.dir) ? 0 : 1;
				int tmp = visited[z.a][z.b][z.dir] + cost;

				if (tmp < visited[x1][x2][newDir])
				{
					visited[x1][x2][newDir] = tmp;
					push(x1,x2,newDir);
				}
			}
		}
	}
	return;
}
int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
	cin>>t;
	for (int test = 1; test <= t; test++)
	{
		front = 0;
		rear = -1;
		cin>>n>>m;
		int st1,st2,ed1,ed2;
		cin>>st2>>st1>>ed2>>ed1;
		st2--;
		st1--;
		ed2--;
		ed1--;
		for (int i = 0; i < m; i++)
		{
			for (int j = 0; j < n; j++)
			{
				cin>>a[i][j];
				visited[i][j][0] = 10000000;
				visited[i][j][1] = 10000000;
			}
		}
		bfs(st1,st2,ed1,ed2);
		int ans = min(visited[ed1][ed2][0], visited[ed1][ed2][1]);
		if (ans == 10000000) cout<<-1<<endl;
		else cout<<ans<<endl;
	}
    return 0;
}
