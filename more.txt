#include <iostream>

using namespace std;
char a[205][205];
int visited[205][205];
int direction[205][205];
int d1[4] = {0,0,-1,1};
int d2[4] = {1,-1,0,0};
int n,m;
int ans = 0;
struct point
{
	int a;
	int b;
};
point q[10000005];
int front = 0;
int rear = -1;
void push(int i,int j)
{
	rear++;
	q[rear].a = i;
	q[rear].b = j;
}
point pop()
{
	return q[front++];
}
bool empty()
{
	return front > rear;
}
void bfs(int i1, int j1, int i2, int j2)
{
	push(i1,j1);
	visited[i1][j1] = 0;
	while (!empty())
	{
		point z = pop();
		for (int i = 0; i<4; i++)
		{
			int x1 = z.a + d1[i];
			int x2 = z.b + d2[i];
			if (x1 < 0 || x1 >= m || x2 < 0|| x2 >= n) continue;
			if (a[x1][x2] == '0')
			{
				int tmp = 0;
				if (direction[z.a][z.b] == 0)
				{
					if (d1[i] != 0) direction[x1][x2] = 1;
					else direction[x1][x2] = 2;
					tmp = visited[z.a][z.b];
				}
				else if (direction[z.a][z.b] != 0)
				{
					if (direction[z.a][z.b] == 1 && d1[i] == 0)
					{
						direction[x1][x2] = 2;
						tmp = visited[z.a][z.b] + 1;
					}
					else if (direction[z.a][z.b] == 2 && d2[i] == 0)
					{
						direction[x1][x2] = 1;
						tmp = visited[z.a][z.b] + 1;
					}
					else
					{
						tmp = visited[z.a][z.b];
						direction[x1][x2] = direction[z.a][z.b];
					}
				}
				if (tmp < visited[x1][x2])
				{
					visited[x1][x2] = tmp;
					push(x1,x2);
				}
			}
		}
	}
	return;
}
int main()
{
	//freopen("Text.txt", "r", stdin);
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
	cin>>t;
	for (int test = 1; test <= t; test++)
	{
		front = 0;
		rear = -1;
		cin>>n>>m;
		int st1,st2,ed1,ed2;
		cin>>st2>>st1>>ed2>>ed1;
		st2--;
		st1--;
		ed2--;
		ed1--;
		for (int i = 0; i < m; i++)
		{
			for (int j = 0; j < n; j++)
			{
				cin>>a[i][j];
				visited[i][j] = 10000000;
				direction[i][j] = 0;
			}
		}
		bfs(st1,st2,ed1,ed2);
		for (int i = 0; i < m; i++)
		{
			for (int j = 0; j < n; j++)
			{
				if (visited[i][j] == 10000000) cout <<"."<<" ";
				else cout<<visited[i][j]<<" ";
			}
			cout<<endl;
		}
		if (visited[ed1][ed2] == 10000000) cout<<-1<<endl;
		else cout<<visited[ed1][ed2]<<endl;
	}
    return 0;
}
