#include <iostream>
using namespace std;
int a[305][305] = {};
int visited[305] = {};
int checked[305] = {};
int d1[4] = {0,0,1,-1};
int d2[4] = {1,-1,0,0};
int ans;
int res1,res2;
int n,g;
struct point
{
	int x;
	int y;
};
int q[1000005] = {};
int rear = -1;
int front = 0;
void push (int x)
{
	rear++;
	q[rear] = x;
}
int pop()
{
	return q[front++];
}
bool empty()
{
	return front > rear;
}
int bfs()
{
	int cnt = 0;
	while(!empty())
	{
		int z = pop();
		checked[z] = 1;
		for (int j = 0; j < n; j++)
		{
			if (checked[j] == 1) continue;
			if (a[z][j] == 0) continue;
			visited[j]++;
			visited[z]++;
			cnt++;
			push(j);
		}
	}
	return cnt;
}
int main() {
	//freopen("Text.txt","r",stdin);
    int t;
    cin >> t;
    for (int test = 1; test <= t; test++) {
		front = 0;
		rear = -1;
		ans = 0;
		res1 = res2 = 0;
		cin>>n;
		for (int i = 0; i < n; i++)
		{
			for (int j = 0; j < n; j++)
			{
				cin>>a[i][j];
			}
			checked[i] = 0;
		}
		for (int i = 0; i < n; i++)
		{
			if (!checked[i])
			{
				front = 0;
				rear = -1;
				for (int j = 0; j < n; j++)
				{
					visited[j] = 0;
				}
				push(i);
				ans++;
				if (bfs() > 0)
				{
					for (int j = 0; j < n; j++)
					{
						if (visited[j] == 1 && j != i) res1++;
					}
				}
				else
				{
					res2++;
				}
			}
		}
		cout<<ans<<" "<<res2<<" "<<res1<<endl;
    }
    return 0;
}
