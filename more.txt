#include <iostream>
using namespace std;
double x[15], mass[15];
int n;

double forceDiff(double p)
{
	double left = 0, right = 0;
	for (int i = 0; i < n; i++)
	{
		double d = p - x[i];
		if (d > 0) left += mass[i] / (d*d);
		else if (d < 0) right += mass[i] / (d*d);
	}
	return left - right;
}

int main()
{
	int t = 10;
	for (int tc = 1; tc <= t; tc++)
	{
		cin >> n;
		for (int i = 0; i < n; i++) cin >> x[i];
		for (int i = 0; i < n; i++) cin >> mass[i];

		// sort theo x tăng dần (kèm mass), bubble/selection sort thủ công
		for (int i = 0; i < n-1; i++)
			for (int j = 0; j < n-1-i; j++)
				if (x[j] > x[j+1])
				{
					double t1=x[j]; x[j]=x[j+1]; x[j+1]=t1;
					double t2=mass[j]; mass[j]=mass[j+1]; mass[j+1]=t2;
				}

		cout << "#" << tc << " ";
		for (int i = 0; i < n-1; i++)
		{
			double lo = x[i], hi = x[i+1];
			for (int iter = 0; iter < 200; iter++)
			{
				double mid = (lo+hi)/2;
				if (forceDiff(mid) > 0) lo = mid;   // left lớn hơn -> điểm cân bằng dịch phải
				else hi = mid;
			}
			cout.precision(10);
			cout << fixed << (lo+hi)/2;
			if (i != n-2) cout << " ";
		}
		cout << endl;
	}
	return 0;
}
