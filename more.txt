#include <iostream>
using namespace std;

int a[105][105];
int n, m;
int dx[] = {0, 0, -1, 1};
int dy[] = {-1, 1, 0, 0};

struct point { int a, b; };
point q[1000005];
int front_, rear_;

void push(int x, int y) { rear_++; q[rear_].a = x; q[rear_].b = y; }
point pop() { return q[front_++]; }
bool empty_() { return front_ > rear_; }

int visited[105][105];
int dis[105][105];

void bfs(int sx, int sy) {
    front_ = 0; rear_ = -1;
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++) {
            visited[i][j] = 0;
            dis[i][j] = -1;
        }
    visited[sx][sy] = 1;
    dis[sx][sy] = 0;
    push(sx, sy);
    while (!empty_()) {
        point z = pop();
        for (int i = 0; i < 4; i++) {
            int x = z.a + dx[i];
            int y = z.b + dy[i];
            if (x < 0 || y < 0 || x >= n || y >= m) continue;
            if (a[x][y] == 2) continue;
            if (visited[x][y]) continue;
            visited[x][y] = 1;
            dis[x][y] = dis[z.a][z.b] + 1;
            push(x, y);
        }
    }
}

const int INF = 1e9;
int dist_[11][11];
int used[10];
int k;
int best;
void solve(int pos, int visitedCount, int sum) {
    if (sum >= best) return; // cắt nhánh: đã tệ hơn best rồi thì bỏ
    if (visitedCount == k) {
        best = sum;
        return;
    }
    for (int j = 0; j < k; j++) {
        if (used[j]) continue;
        if (dist_[pos][j + 1] == -1) continue; // không tới được
        used[j] = 1;
        solve(j + 1, visitedCount + 1, sum + dist_[pos][j + 1]);
        used[j] = 0;
    }
}

int main() {
    int t;
    cin >> t;
    for (int test = 1; test <= t; test++) {
        cin >> n >> m;
        int st1 = -1, st2 = -1;
        int dirtyR[15], dirtyC[15];
        k = 0;

        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++) {
                cin >> a[i][j];
                if (a[i][j] == 3) { st1 = i; st2 = j; }
                if (a[i][j] == 1) { dirtyR[k] = i; dirtyC[k] = j; k++; }
            }

        if (k == 0) {
            cout << "Case #" << test << "\n" << 0 << "\n";
            continue;
        }

        bfs(st1, st2);
        for (int i = 0; i < k; i++) dist_[0][i + 1] = dis[dirtyR[i]][dirtyC[i]];

        for (int i = 0; i < k; i++) {
            bfs(dirtyR[i], dirtyC[i]);
            dist_[i + 1][0] = dis[st1][st2];
            for (int j = 0; j < k; j++)
                dist_[i + 1][j + 1] = dis[dirtyR[j]][dirtyC[j]];
        }

        bool impossible = false;
        for (int i = 0; i < k; i++)
            if (dist_[0][i + 1] == -1) impossible = true;

        cout << "Case #" << test << "\n";
        if (impossible) {
            cout << -1 << "\n";
            continue;
        }

        for (int i = 0; i < k; i++) used[i] = 0;
        best = INF;
        solve(0, 0, 0);

        cout << (best >= INF ? -1 : best) << "\n";
    }
    return 0;
}
