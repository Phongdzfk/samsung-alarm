#include <iostream>
using namespace std;
int n;
int a[105][105];
int visited[105];
int q[105];
int front_, rear_;

void push(int x){ q[++rear_] = x; }
int pop(){ return q[front_++]; }
bool empty_(){ return front_ > rear_; }

int demThanhPhan(int bo)
{
	for (int i = 1; i <= n; i++) visited[i] = 0;
	int cnt = 0;
	for (int i = 1; i <= n; i++)
	{
		if (i == bo) continue;
		if (visited[i]) continue;
		cnt++;
		front_ = 0; rear_ = -1;
		push(i);
		visited[i] = 1;
		while (!empty_())
		{
			int z = pop();
			for (int j = 1; j <= n; j++)
			{
				if (j == bo) continue;
				if (a[z][j] == 1 && visited[j] == 0)
				{
					visited[j] = 1;
					push(j);
				}
			}
		}
	}
	return cnt;
}

int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	int t;
	cin >> t;
	while (t--)
	{
		cin >> n;
		for (int i = 1; i <= n; i++)
			for (int j = 1; j <= n; j++)
				cin >> a[i][j];

		int goc = demThanhPhan(0);   // 0 = khong bo dao nao

		int ans = 0;
		int maxPhan = goc;
		for (int i = 1; i <= n; i++)
		{
			int tmp = demThanhPhan(i);
			if (tmp > maxPhan)
			{
				maxPhan = tmp;
				ans = i;
			}
		}
		cout << ans << endl;
	}
	return 0;
}
