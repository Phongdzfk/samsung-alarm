#include <iostream>
using namespace std;
long long a[105][105];
long long visited[105][105];
long long checked[105][105];
int dx[8] = {0,0,1,-1,1,1,-1,-1};
int dy[8] = {1,-1,0,0,1,-1,1,-1};
int ans;
int f;
long long n,m = -1;

struct point{ int x,y; };
point q[9000005];
int front, rear;
void push(int x,int y){ rear++; q[rear].x=x; q[rear].y=y; }
point pop(){ return q[front++]; }
bool empty(){ return front > rear; }

void lanbien()
{
	while(!empty())
	{
		point z = pop();
		for (int i = 0; i < 8; i++)
		{
			int x = z.x + dx[i];
			int y = z.y + dy[i];
			if (x < 0 || y < 0 || x > n + 1 || y > m + 1) continue;
			if (visited[x][y]) continue;
			if (f >= a[x][y])
			{
				visited[x][y] = 1;
				push(x,y);
			}
		}
	}
}
void checkk()
{
	while(!empty())
	{
		point z = pop();
		for (int i = 0; i < 8; i++)
		{
			int x = z.x + dx[i];
			int y = z.y + dy[i];
			if (x <= 0 || y <= 0 || x > n || y > m) continue;
			if (checked[x][y] || visited[x][y]) continue;
			checked[x][y] = 1;
			push(x,y);
		}
	}
}
int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	int cnt = 0;
	//freopen("Text.txt","r",stdin);
	while (m != 0 || n != 0)
	{
		ans = -1;
		cnt++;
		cin>>m>>n;
		front = 0;
		rear = -1;
		for (int i = 1; i <= n; i++)
		{
			for (int j = 1; j <= m; j++)
			{
				cin>>a[i][j];
			}
		}
		for (int z = 0; z < 1001; z++)
		{
			front = 0;
			rear = -1;
			f = z;
			for (int i = 0; i <= n + 1; i++)
			{
				for (int j = 0; j <= m + 1; j++)
				{
					visited[i][j] = 0;
					checked[i][j] = 0;
				}
			}
			push(0,0);
			visited[0][0] = 1;
			lanbien();
			front = 0;
			rear = -1;
			int test = 0;
			for (int i = 1; i <= n; i++)
			{
				for (int j = 1; j <= m; j++)
				{
					if (visited[i][j] == 0)
					{
						test = 1;
						push(i,j);
						checked[i][j] = 1;
						break;
					}
				}
				if (test == 1) break;
			}
			checkk();
			int fail = 0;
			for (int i = 1; i <=n; i++)
			{
				for (int j = 1; j <= m; j++)
				{
					if (visited[i][j] != 1 && checked[i][j] == 0) 
					{
						fail = 1;
						break;
					}
				}
				if (fail == 1) break;
			}
			if (fail == 1) ans = z;
			if (ans != -1) break;
		}
		if (ans == -1) cout<<"Case "<<cnt<<": "<<"Island never splits."<<endl;
		else cout<<"Case "<<cnt<<": "<<"Island splits when ocean rises "<<ans<<" feet."<<endl;
	}
	return 0;
}
