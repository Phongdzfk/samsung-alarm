#include<iostream>
using namespace std;

int heap[1005];
int heapSize;

void heapPush(int x)
{
	heap[++heapSize] = x;
	int c = heapSize;
	while (c > 1)
	{
		int p = c / 2;
		if (heap[p] <= heap[c]) break;
		int tmp = heap[p]; heap[p] = heap[c]; heap[c] = tmp;
		c = p;
	}
}

int heapPop()
{
	int top = heap[1];
	heap[1] = heap[heapSize--];
	int p = 1;
	while (true)
	{
		int l = p*2, r = p*2+1, small = p;
		if (l <= heapSize && heap[l] < heap[small]) small = l;
		if (r <= heapSize && heap[r] < heap[small]) small = r;
		if (small == p) break;
		int tmp = heap[p]; heap[p] = heap[small]; heap[small] = tmp;
		p = small;
	}
	return top;
}

int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);

	int t;
	cin>>t;
	for (int test_case = 1; test_case <= t; ++test_case)
	{
		int n;
		cin>>n;
		heapSize = 0;
		for (int i = 0; i < n; i++)
		{
			int s;
			cin>>s;
			heapPush(s);
		}

		long long ans = 0;
		while (heapSize > 1)
		{
			int a = heapPop();
			int b = heapPop();
			int merged = a + b;
			ans += merged;
			heapPush(merged);
		}

		cout<<"Case #"<<test_case<<endl<<ans<<endl;
	}
	return 0;
}
