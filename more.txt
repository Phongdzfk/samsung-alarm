#include<iostream>
using namespace std;
int m;
int a[105][105] = {};
int d[105] = {};
int parent[105] = {};
int v[105] = {};
int cnt,ans;
int check = 0;
void dfs(int z)
{
	v[z] = 1;
	for (int i = 0; i < m; i++)
	{
		if (v[i] && parent[z] != i && z != i && a[z][i] == 1)
		{
			parent[i] = z;
			//cout<<z<<" "<<i<<endl;
			int tmp = z;
			cnt = d[i] + d[z];
			//int idx1 = i;
			//int idx2 = z;
			a[i][z] = 0;
			a[z][i] = 0;
			while (tmp != i)
			{
				if (d[tmp] + d[parent[tmp]] < cnt)
				{
					cnt = d[tmp] + d[parent[tmp]];
					//idx1 = tmp;
					//idx2 = parent[tmp];
				}
				a[tmp][parent[tmp]] = 0;
				a[parent[tmp]][tmp] = 0;
				tmp = parent[tmp];
			}
			//a[idx1][idx2] = 0;
			//a[idx2][idx1] = 0;
			ans+=cnt;
		}
		else if (!v[i] && a[z][i] == 1)
		{
			//cout<<z<<" noi "<<i<<endl;
			parent[i] = z;
			dfs(i);
		}
	}
	return;
}
int main()
{
	int t;
	cin>>t;
	for(int test_case = 1; test_case <= t; ++test_case)
	{
		cnt = 0;
		ans = 0;
		cin>>m;
		for (int i = 0; i < m; i++)
		{
			for (int j = 0; j < m; j++)
			{
				a[i][j] = 0;
			}
			v[i] = 0;
			parent[i] = 10000;
		}
		for (int i = 0; i < m; i++)
		{
			int i1,n;
			cin>>i1>>d[i]>>n;
			for (int j = 0; j < n; j++)
			{
				int tmp;
				cin>>tmp;
				a[i][tmp] = 1;
			}
		}
		for (int i = 0; i < m; i++)
		{
			if (!v[i])
			{
				check = 1;
				dfs(i);
			}
		}
		cout<<"Case #"<<test_case<<endl<<ans<<endl;
	}
	return 0;
}
