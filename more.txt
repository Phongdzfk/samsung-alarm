#include <iostream>
using namespace std;
int n;
int num[25];
int cost[25];
int choose[25];
int fatig[25];
int ans;

void quaylui(int cong, int total, int binhsi)
{
	if (total >= ans) return;
	if (cong == n+1)
	{
		if (total < ans) ans = total;
		return;
	}

	quaylui(cong+1, total + cost[cong], binhsi);

	choose[cong] = num[cong];
	fatig[cong] = 3;
	quaylui(cong+1, total + cost[cong]*2, binhsi + num[cong]);
	choose[cong] = 0;
	fatig[cong] = 0;

	if (binhsi >= num[cong])
	{
		int opp = num[cong];
		int saveChoose[25];
		int saveFatig[25];
		for (int i = 1; i <= n; i++)
		{
			saveChoose[i] = choose[i];
			saveFatig[i] = fatig[i];
		}

		int kill = opp;
		for (int i = 1; i <= n && kill > 0; i++)
		{
			if (choose[i] > 0)
			{
				if (choose[i] <= kill) { kill -= choose[i]; choose[i] = 0; }
				else { choose[i] -= kill; kill = 0; }
			}
		}

		for (int i = 1; i <= n; i++)
			if (choose[i] > 0)
			{
				fatig[i] -= 1;
				if (fatig[i] == 0) choose[i] = 0;
			}

		quaylui(cong+1, total, binhsi - opp);

		for (int i = 1; i <= n; i++)
		{
			choose[i] = saveChoose[i];
			fatig[i] = saveFatig[i];
		}
	}
}

int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	int t;
	cin>>t;
	for (int test=1; test<=t; test++)
	{
		ans = 1e9;
		cin>>n;
		for (int i = 1; i <= n; i++)
		{
			cin>>num[i]>>cost[i];
			choose[i] = 0;
			fatig[i] = 0;
		}
		quaylui(1, 0, 0);
		cout<<"#"<<test<<" "<<ans<<endl;
	}
	return 0;
}
