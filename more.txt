#include <iostream>
using namespace std;
int vang[105][105] = {};
int a[105][105] = {};
int thutuy[5] = {};
int thutux[5] = {};
int num[105][105] = {};
int visited[105][105] = {};
int checked[105][105] = {};
int d1[4] = {0,0,1,-1};
int d2[4] = {1,-1,0,0};
int ans;
int n,g;
struct point
{
	int x;
	int y;
};
point q[1000005] = {};
int rear = -1;
int front = 0;
void push (int x, int y)
{
	rear++;
	q[rear].x = x;
	q[rear].y = y;
}
point pop()
{
	return q[front++];
}
bool empty()
{
	return front > rear;
}
void bfs()
{
	while(!empty())
	{
		point z = pop();
		for (int i = 0; i < 4; i++)
		{
			int x = z.x + d1[i];
			int y = z.y + d2[i];
			if (x <= 0 || x > n || y <= 0 || y > n) continue;
			if (a[x][y] == 0) continue;
			if (checked[x][y] != -1 ) continue;
			num[x][y]++;
			checked[x][y] = checked[z.x][z.y] + 1;
			push(x,y);
		}
	}
	return;
}
int main() {
	//freopen("Text.txt","r",stdin);
    int t;
    cin >> t;
    for (int test = 1; test <= t; test++) {
		front = 0;
		rear = -1;
		ans = 1e9;
		cin>>n>>g;
		for (int i = 1; i <=n; i++)
		{
			for (int j = 1; j <=n; j++)
			{
				vang[i][j] = 0;
			}
		}
		for (int i = 0; i < g; i++)
		{
			int x,y;
			cin>>x>>y;
			thutux[i+1] = x;
			thutuy[i+1] = y;
			vang[x][y] = 1;
		}
		for (int i = 1; i <=n; i++)
		{
			for (int j = 1; j <=n; j++)
			{
				cin>>a[i][j];
				visited[i][j] = -1;
				num[i][j] = 0;
			}
		}
		for (int i = 1; i <=n; i++)
		{
			for (int j = 1; j <=n; j++)
			{
				if (vang[i][j] == 1) 
				{
					for (int i1 = 1; i1 <=n; i1++)
					{
						for (int j1 = 1; j1 <=n; j1++)
						{
							checked[i1][j1] = -1;
						}
					}
					front = 0;
					rear = -1;
					push(i,j);
					checked[i][j] = 0;
					bfs();
					for (int i1 = 1; i1 <=n; i1++)
					{
						for (int j1 = 1; j1 <=n; j1++)
						{
							visited[i1][j1] = max(visited[i1][j1],checked[i1][j1]);
						}
					}
				}
			}
		}
		int curnum = 0;
		for (int i = 1; i <=n; i++)
		{
			for (int j = 1; j <=n; j++)
			{

				if (num[i][j] > curnum) curnum = num[i][j];
			}
		}
		int idx1 = 0;
		int idx2 = 0;
		for (int i = 1; i <=n; i++)
		{
			for (int j = 1; j <=n; j++)
			{
				if (num[i][j] == curnum) 
				{
					if (visited[i][j] < ans && visited[i][j] != 0) 
					{
						ans = visited[i][j];
						idx1 = i;
						idx2 = j;
					}
				}
			}
		}
		if (ans == 1e9) ans = -1;
		cout<<"Case #"<<test<<endl<<ans<<endl;
		if (ans != -1)
		{
			for (int i1 = 1; i1 <=n; i1++)
			{
				for (int j1 = 1; j1 <=n; j1++)
				{
					checked[i1][j1] = -1;
				}
			}
			front = 0;
			rear = -1;
			push(idx1,idx2);
			checked[idx1][idx2] = 0;
			bfs();
			for (int i = 1; i <= g; i++)
			{
				if (vang[thutux[i]][thutuy[i]] == 1 && checked[thutux[i]][thutuy[i]] == -1) 
					{
						cout<<thutux[i]<<" "<<thutuy[i]<<endl;
					}
			}
		}
    }
    return 0;
}
