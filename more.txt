#include <iostream>
using namespace std;
int a[3005][3005] = {};
int n,m,p,q,s,t;
int d1[4] = {-1,1,-1,1};
int d2[4] = {-1,1,1,-1};
int visited[3005][3005] = {};
int rear = -1;
int front = 0;
struct point
{
	int x;
	int y;
};
point qu[100000005];
void push (int x, int y)
{
	rear++;
	qu[rear].x = x;
	qu[rear].y = y;
}
point pop()
{
	return qu[front++];
}
bool empty()
{
	return front > rear;
}
void bfs()
{
	while(!empty())
	{
		point z = pop();
		for (int i = 0; i < 4; i++)
		{
			int x1 = z.x + d1[i];
			int x2 = z.y + d2[i];
			while (x1 >= 1 && x1 <=n && x2 >= 1 && x2 <= n && a[x1][x2] != 1)
			{
				if (visited[x1][x2] == -1)
				{
					visited[x1][x2] = visited[z.x][z.y] + 1;
					push(x1,x2);
					x1 += d1[i];
					x2 += d2[i];
				}
				else 
				{
					x1 += d1[i];
					x2 += d2[i];
				}
				
			}
		}
	}
	return;
}

int main()
{
	freopen("Text.txt","r",stdin);
	int t;
	cin>>t;
	for (int test = 1; test <= t; test++)
	{
		front = 0;
		rear = -1;
		cin>>n>>m>>p>>q>>s>>t;
		for (int i = 1; i <= n; i++)
		{
			for (int j = 1; j <= n; j++)
			{
				a[i][j] = 0;
				visited[i][j] = -1;
			}
		}
		for (int i = 1; i <= m; i++)
		{
			int x,y;
			cin>>x>>y;
			a[x][y] = 1;
		}
		push(p,q);
		visited[p][q] = 0;
		bfs();
		cout<<visited[s][t]<<endl;
	}
	return 0;
}
