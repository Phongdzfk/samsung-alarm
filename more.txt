#include <iostream>
#include <iomanip>
using namespace std;
int n;
int num[25];
int cost[25];
int choose[25];
int fatig[25];
int ans;

void quaylui(int cong, int total, int binhsi)
{
	if (total >= ans) return;
	if (cong == n+1)
	{
		if (total < ans) ans = total;
		return;
	}

	quaylui(cong+1, total + cost[cong], binhsi);

	choose[cong] = num[cong];
	fatig[cong] = 3;
	quaylui(cong+1, total + cost[cong]*2, binhsi + num[cong]);
	choose[cong] = 0;
	fatig[cong] = 0;

	if (binhsi >= num[cong])
	{
		int opp = num[cong];
		int saveChoose[25];
		int saveFatig[25];
		for (int i = 1; i <= n; i++)
		{
			saveChoose[i] = choose[i];
			saveFatig[i] = fatig[i];
		}

		int kill = opp;
		for (int i = 1; i <= n && kill > 0; i++)
		{
			if (choose[i] > 0)
			{
				if (choose[i] <= kill) { kill -= choose[i]; choose[i] = 0; }
				else { choose[i] -= kill; kill = 0; }
			}
		}

		for (int i = 1; i <= n; i++)
			if (choose[i] > 0)
			{
				fatig[i] -= 1;
				if (fatig[i] == 0)
				{
					binhsi-=choose[i];
					choose[i] = 0;
				}
			}

		quaylui(cong+1, total, binhsi - opp);

		for (int i = 1; i <= n; i++)
		{
			choose[i] = saveChoose[i];
			fatig[i] = saveFatig[i];
		}
	}
}

int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	int t;
	cin>>t;
	for (int test=1; test<=t; test++)
	{
		ans = 1e9;
		cin>>n;
		for (int i = 1; i <= n; i++)
		{
			cin>>num[i]>>cost[i];
			choose[i] = 0;
			fatig[i] = 0;
		}
		quaylui(1, 0, 0);
		cout<<"#"<<test<<" "<<ans<<endl;
	}
	return 0;
}

void quaylui(int loai,int energy,int thoigian,int km)
{
	if (energy < 0) return;
	if (thoigian >= ans) return;
	if (loai == 5)
	{
		int conlai = d - km;
		int e = energy - mana[5] *km;
		int tg = thoigian + tt[5]*km;
		if (e > 0 && tg < ans) ans = tg;
		return;
	}
	for (int i = 0; i <= km; i++)
	{
		quaylui(loai+1,energy-(mana[loai] * i),thoigian+(tt[loai] * i),km - i);
	}
}

#include <iostream>
using namespace std;
int n,e,k,t;
float ans1;
int ans2;
float a[105][105];
int front_ = 0;
int rear_ = -1;
float xacsuat[105][1105];
bool visited[105][1105];

struct point
{
	int node;
	int time;
};
point q[300005];
void push(int x,int tm)
{
	rear_++;
	q[rear_].node = x;
	q[rear_].time = tm;
}
point pop()
{
	return q[front_++];
}
bool empty_()
{
	return front_ > rear_;
}

void bfs(int start,int startTime)
{
	front_ = 0; rear_ = -1;
	xacsuat[start][startTime] = 1.0;
	push(start,startTime);
	visited[start][startTime] = true;

	while(!empty_())
	{
		point z = pop();
		int sr = z.node;
		int tm = z.time;
		if (tm > t) continue;

		for (int i = 1; i <= n; i++)
		{
			if (a[sr][i] != -1)
			{
				int nt = tm + 10;
				xacsuat[i][nt] += xacsuat[sr][tm] * a[sr][i];
				if (!visited[i][nt])
				{
					visited[i][nt] = true;
					push(i,nt);
				}
			}
		}
	}
}

int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	//freopen("Text.txt","r",stdin);
	for (int test=1; test<=10; test++)
	{
		if(!(cin>>n>>e>>k>>t)) break;
		for (int i = 1; i <= n; i++)
		{
			for (int j = 1; j <= n; j++)
			{
				a[i][j] = -1;
			}
		}
		for (int i = 1; i <= e; i++)
		{
			int source,des; float value;
			cin>>source>>des>>value;
			a[source][des] = value;
		}

		for (int i = 1; i <= n; i++)
		{
			for (int j = 0; j <= t+15; j++)
			{ 
				xacsuat[i][j] = 0; visited[i][j] = false; 
			}
		}
		bfs(1,0);
		ans1 = 0.0; ans2 = 0;
		int moc = (t/10)*10;
		for (int i = 1; i <= n; i++)
		{
			if (xacsuat[i][moc] > ans1 + 1e-12)
			{ 
				ans1 = xacsuat[i][moc]; ans2 = i; 
			}
		}
		if (ans2 == 0) ans1 = 0.0;
		cout.precision(6);
		cout<<"#"<<test<<" "<<ans2<<" "<<fixed<<ans1<<" ";

		for (int i = 1; i <= n; i++)
		{
			for (int j = 0; j <= t+15; j++)
			{ 
				xacsuat[i][j] = 0; visited[i][j] = false; 
			}
		}
		bfs(1,k);
		ans1 = 0.0; ans2 = 0;
		int moc2 = k + ((t-k)/10)*10;
		for (int i = 1; i <= n; i++)
		{
			if (xacsuat[i][moc2] > ans1 + 1e-12)
			{ 
				ans1 = xacsuat[i][moc2]; ans2 = i; 
			}
		}
		if (ans2 == 0) ans1 = 0.0;
		cout.precision(6);
		cout<<ans2<<" "<<fixed<<ans1<<endl;
	}
	return 0;
}
