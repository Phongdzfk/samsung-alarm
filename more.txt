#include <iostream>
using namespace std;
int n;
int front = 0;
int rear = -1;
int cua[3];
int ans;
int vitricua[3];
int chongoi[100];
int visited[5];
int dx[4] = {0,0,1,-1};
int dy[4] = {1,-1,0,0};
struct point
{
	int a;
	int b;
};
point q[1000005];
void push(int x, int y)
{
	rear++;
	q[rear].a = x;
	q[rear].b = y;
}
point pop()
{
	return q[front++];
}
bool empty()
{
	return front > rear;
}
int khachlen(int x, int pri)
{
	int cnt = 0;
	int acc = 1;
	int sokhach = cua[x];
	int vitri = vitricua[x];
	if (chongoi[vitri] == 0) 
	{
		chongoi[vitri] = 1;
		sokhach -= 1;
		cnt+=acc;
	}
	int l = vitri - 1;
	int r = vitri + 1;
	while(sokhach > 1)
	{
		acc++;
		if (l >=1 && chongoi[l] == 0)
		{
			chongoi[l] = 1;
			cnt+=acc;
			sokhach--;
		}
		if (r <= n && chongoi[r] == 0)
		{
			chongoi[r] = 1;
			cnt+=acc;
			sokhach--;
		}
		l--;
		r++;
		if (l < 1 && r > n) break;
	}
	if (sokhach > 0)
	{
		acc++;
		int accl;
		int accr;
		accl = accr = acc;
		while (r <= n && chongoi[r] == 1) 
		{
			r++;
			accr++;
		}
		while (l >= 1 && chongoi[l] == 1) 
		{
			l--;
			accl++;
		}
		if (l < 1) 
		{
			chongoi[r] = 1;
			cnt+=accr;
		}
		else if ( r> n)
		{
			chongoi[l] = 1;
			cnt+=accl;
		}
		else if (l >= 1 && r <=n)
		{
			if (accl > accr)
			{
				chongoi[r] = 1;
				cnt+=accr;
			}
			else if (accl < accr)
			{
				chongoi[l] = 1;
				cnt+=accl;
			}
			else if (accl == accr)
			{
				if (pri == 0)
				{
					chongoi[l] = 1;
					cnt+=accl;
				}
				else if (pri == 1)
				{
					chongoi[r] = 1;
					cnt+=accr;
				}
			}
		}
	}
	return cnt;
}
void quaylui(int pri,int total)
{
	if (total > ans) return;
	for (int i = 1; i <=3; i++)
	{
		if (visited[i] == 0)
		{
			visited[i] = 1;
			int saved[100] = {};
			for (int x = 1; x <=n; x++)
			{
				saved[x] = chongoi[x];
			}
			int tmp = khachlen(i,pri);
			cout<<"cong "<<i<<" checktmp "<<tmp<<endl;
			quaylui(0,total + tmp);
			quaylui(1,total + tmp);
			visited[i] = 0;
			for (int x = 1; x <= n; x++)
			{
				chongoi[x] = saved[x];
			}
		}
	}
	if (total < ans) ans = total;
	return;
}
int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	//freopen("Text.txt","r",stdin);
	int t;
	cin>>t;
	for (int test=1; test<=t; test++)
	{
		ans = 1e9;
		cin>>n;
		for (int i = 1; i <=n; i++)
		{
			chongoi[i] = 0;
		}
		for (int i = 1; i <=3; i++)
		{
			int p,c;
			cin>>p>>c;
			vitricua[i] = p;
			cua[i] = c;
			visited[i] = 0;
		}
		quaylui(0,0);
		quaylui(1,0);
		cout<<"Case #"<<test<<endl<<ans<<endl;
	}
	return 0;
}
