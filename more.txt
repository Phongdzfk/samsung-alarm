#include <iostream>
using namespace std;
int n,e,k,t;
double ans1;
int ans2;
double a[105][105];
int front_ = 0;
int rear_ = -1;
double xacsuat[105][1105];
bool inq[105][1105];

struct point
{
	int node;
	int time;
};
point q[300005];
void push(int x,int tm)
{
	rear_++;
	q[rear_].node = x;
	q[rear_].time = tm;
}
point pop()
{
	return q[front_++];
}
bool empty_()
{
	return front_ > rear_;
}

void bfs(int start,int startTime)
{
	front_ = 0; rear_ = -1;
	xacsuat[start][startTime] = 1.0;
	push(start,startTime);
	inq[start][startTime] = true;

	while(!empty_())
	{
		point z = pop();
		int sr = z.node;
		int tm = z.time;
		if (tm > t) continue;

		for (int i = 1; i <= n; i++)
		{
			if (a[sr][i] != -1)
			{
				int nt = tm + 10;
				if (nt > t + 10) continue;
				xacsuat[i][nt] += xacsuat[sr][tm] * a[sr][i];
				if (!inq[i][nt])
				{
					inq[i][nt] = true;
					push(i,nt);
				}
			}
		}
	}
}

{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	for (int test=1; test<=10; test++)
	{
		if(!(cin>>n>>e>>k>>t)) break;
		for (int i = 1; i <= n; i++)
			for (int j = 1; j <= n; j++)
				a[i][j] = -1;
		for (int i = 1; i <= e; i++)
		{
			int source,des; double value;
			cin>>source>>des>>value;
			a[source][des] = value;
		}

		for (int i = 1; i <= n; i++)
			for (int j = 0; j <= t+15; j++)
			{ xacsuat[i][j] = 0; inq[i][j] = false; }
		bfs(1,0);
		ans1 = 0.0; ans2 = 0;
		int moc = (t/10)*10;
		for (int i = 1; i <= n; i++)
			if (xacsuat[i][moc] > ans1 + 1e-12)
			{ ans1 = xacsuat[i][moc]; ans2 = i; }
		if (ans2 == 0) ans1 = 0.0;
		cout.precision(6);
		cout<<"#"<<test<<" "<<ans2<<" "<<fixed<<ans1<<" ";

		for (int i = 1; i <= n; i++)
			for (int j = 0; j <= t+15; j++)
			{ xacsuat[i][j] = 0; inq[i][j] = false; }
		bfs(1,k);
		ans1 = 0.0; ans2 = 0;
		int moc2 = k + ((t-k)/10)*10;
		for (int i = 1; i <= n; i++)
			if (xacsuat[i][moc2] > ans1 + 1e-12)
			{ ans1 = xacsuat[i][moc2]; ans2 = i; }
		if (ans2 == 0) ans1 = 0.0;
