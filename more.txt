
#include <iostream>
using namespace std;
int a[305][305] = {};
int disc[305], low[305], par[305];
int n;
int timer_;
int ans, res1, res2;
bool visitedNode[305];

void dfs(int u)
{
	visitedNode[u] = true;
	disc[u] = low[u] = ++timer_;
	int childCount = 0;

	for (int v = 0; v < n; v++)
	{
		if (a[u][v] == 0) continue;
		if (v == par[u]) continue;   // bỏ qua cạnh vừa đi từ cha xuống (tránh coi cạnh cha-con là back-edge)

		if (visitedNode[v])
		{
			// back-edge: cập nhật low[u]
			if (disc[v] < low[u]) low[u] = disc[v];
		}
		else
		{
			par[v] = u;
			childCount++;
			dfs(v);
			if (low[v] < low[u]) low[u] = low[v];

			if (low[v] > disc[u]) res1++;   // cạnh (u,v) là bridge
		}
	}
}

int main()
{
	int t;
	cin>>t;
	for (int test = 1; test <= t; test++)
	{
		cin>>n;
		for (int i = 0; i < n; i++)
			for (int j = 0; j < n; j++)
				cin>>a[i][j];

		for (int i = 0; i < n; i++)
		{
			visitedNode[i] = false;
			disc[i] = low[i] = 0;
			par[i] = -1;
		}
		timer_ = 0;
		ans = 0; res1 = 0; res2 = 0;

		for (int i = 0; i < n; i++)
		{
			if (!visitedNode[i])
			{
				ans++;
				bool hasEdge = false;
				for (int j = 0; j < n; j++)
					if (a[i][j] != 0) { hasEdge = true; break; }

				dfs(i);

				if (!hasEdge)
				{
					bool isolated = true;
					for (int j = 0; j < n; j++)
						if (a[i][j] != 0) { isolated = false; break; }
					if (isolated) res2++;
				}
			}
		}

		cout<<ans<<" "<<res2<<" "<<res1<<endl;
	}
	return 0;
}
