struct point{ int x,y; };
point q[9000005];
int front_, rear_;
void push(int x,int y){ rear_++; q[rear_].x=x; q[rear_].y=y; }
point pop(){ return q[front_++]; }

void gatcongtac(int x)
{
	for (int i = x; i <= n; i += x+1)
	{
		a[i] = a[i] ^ 1;
	}
}
void quaylui(int x, int cnt)
{
	if (cnt > 3) return;
	if (x == k + 1)
	{
		int tmp = 0;
		//cout<<"socongtacgat "<<cnt<<endl;
		for (int i = 1; i <= n; i++)
		{
			//cout<<a[i]<<" ";
			if (a[i] == 0) tmp++;
		}
		//cout<<endl;
		if (tmp > ans) ans = tmp;
		return;
	}
	quaylui(x+1,cnt);
	gatcongtac(x);
	quaylui(x+1,cnt+1);
	gatcongtac(x);
}
