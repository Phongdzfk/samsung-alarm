#include <iostream>
using namespace std;
int a[105][105];
int n;
int dx[4] = {0,0,1,-1};
int dy[4] = {1,-1,0,0};

struct point{ int x,y; };
point q[10005];
int front_, rear_;
void push(int x,int y){ rear_++; q[rear_].x=x; q[rear_].y=y; }
point pop(){ return q[front_++]; }
bool empty_(){ return front_ > rear_; }

bool visited[105][105];

bool canReach(int lo, int hi)
{
	if (a[0][0] < lo || a[0][0] > hi) return false;
	if (a[n-1][n-1] < lo || a[n-1][n-1] > hi) return false;

	for (int i=0;i<n;i++)
		for (int j=0;j<n;j++)
			visited[i][j] = false;

	front_ = 0; rear_ = -1;
	push(0,0);
	visited[0][0] = true;

	while (!empty_())
	{
		point z = pop();
		if (z.x == n-1 && z.y == n-1) return true;
		for (int k=0;k<4;k++)
		{
			int x1 = z.x+dx[k];
			int y1 = z.y+dy[k];
			if (x1<0||x1>=n||y1<0||y1>=n) continue;
			if (visited[x1][y1]) continue;
			if (a[x1][y1] < lo || a[x1][y1] > hi) continue;
			visited[x1][y1] = true;
			push(x1,y1);
		}
	}
	return false;
}

int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	int t;
	cin>>t;
	for (int test=1; test<=t; test++)
	{
		cin>>n;
		for (int i=0;i<n;i++)
			for (int j=0;j<n;j++)
				cin>>a[i][j];

		int ans = 1000000000;
		for (int lo = 0; lo <= 110; lo++)
		{
			if (lo > a[0][0]) break;   // lo phải <= a[0][0] để A hợp lệ trong [lo,hi], quá ngưỡng thì dừng hẳn

			int l = lo, r = 110;
			int found = -1;
			while (l <= r)
			{
				int mid = (l+r)/2;
				if (canReach(lo, mid))
				{
					found = mid;
					r = mid-1;
				}
				else l = mid+1;
			}
			if (found != -1)
			{
				int range = found - lo;
				if (range < ans) ans = range;
			}
		}
		cout<<"#"<<test<<" "<<ans<<endl;
	}
	return 0;
}
