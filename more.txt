#include <iostream>
using namespace std;

// Hoán đổi giá trị 2 biến int
void swapVal(int &a, int &b) {
    int t = a;
    a = b;
    b = t;
}

// Partition cho quicksort, sắp xếp theo mảng end tăng dần
// (mảng start được hoán đổi song song để giữ đúng cặp start-end)
int partition(int start[], int end[], int low, int high) {
    int pivot = end[high];
    int i = low - 1;

    for (int j = low; j < high; j++) {
        if (end[j] < pivot) {
            i++;
            swapVal(start[i], start[j]);
            swapVal(end[i], end[j]);
        }
    }
    swapVal(start[i + 1], start[high]);
    swapVal(end[i + 1], end[high]);

    return i + 1;
}

void quickSort(int start[], int end[], int low, int high) {
    if (low < high) {
        int pi = partition(start, end, low, high);
        quickSort(start, end, low, pi - 1);
        quickSort(start, end, pi + 1, high);
    }
}

int main() {
    int n;
    cin >> n; // Số lượng cuộc họp

    int *start = new int[n];
    int *end = new int[n];

    for (int i = 0; i < n; i++) {
        cin >> start[i] >> end[i];
    }

    // Bước 1: Sắp xếp theo end_time (thay cho sort() + struct)
    quickSort(start, end, 0, n - 1);

    int count = 0;
    int last_end_time = -1;

    // Bước 2 & 3: Lựa chọn tham lam
    for (int i = 0; i < n; i++) {
        if (start[i] >= last_end_time) {
            count++;
            last_end_time = end[i];
        }
    }

    cout << count << "\n";

    delete[] start;
    delete[] end;

    return 0;
}
