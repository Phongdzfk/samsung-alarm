#include<iostream>
using namespace std;
int n,m;
int st,ed;
int a[305][305] = {};
int dis[305][305] = {};
int v[305][305] = {};
int d1[4] = {0,0,1,-1};
int d2[4] = {1,-1,0,0};
void djk(int si,int sj)
{
	for (int i = 0; i < 4; i++)
	{
		int xi = si + d1[i];
		int xj = sj + d2[i];
		if (xi < 0 || xi >= m || xj < 0 || xj >=n) continue;
		if (a[xi][xj] != -1)
		{
			 dis[xi][xj] = a[xi][xj];
			 //cout<<"ini "<<xi<<" "<<xj<<" "<<dis[xi][xj]<<endl;
		}
	}
	v[si][sj] = 1;
	int cnt = 1;
	while (cnt < n * m)
	{
		int minn = 1000000;
		int idx1,idx2 = -1;
		for (int z1 = 0; z1 < m; z1++)
		{
			for (int z2 = 0; z2 < n; z2++)
			{
				if (!v[z1][z2] && dis[z1][z2] < minn)
				{
					minn = dis[z1][z2];
					idx1 = z1;
					idx2 = z2;
				}
			}
		}
		//cout<<"find "<<minn<<" "<<idx1<<" "<<idx2<<endl;
		if (idx1 == -1) return;
		v[idx1][idx2] = 1;
		if (idx1 == st && idx2 == ed) return;
		for (int i = 0; i < 4; i++)
		{
			int xi = idx1 + d1[i];
			int xj = idx2 + d2[i];
			if (xi < 0 || xi >= m || xj < 0 || xj >=n) continue;
			if (!v[xi][xj] && a[xi][xj] != -1 && dis[xi][xj] > dis[idx1][idx2] + a[xi][xj])
			{
				dis[xi][xj] = dis[idx1][idx2] + a[xi][xj];
				//cout<<"upd "<<xi<<" "<<xj<<" "<<dis[xi][xj]<<endl;
			}
		}
		cnt++;
	}
	return;
}
int main()
{
	int t;
	cin>>t;
	for(int test_case = 1; test_case <= t; ++test_case)
	{
		st = ed = 0;
		cin>>m>>n;
		for (int i = 0; i < m; i++)
		{
			for (int j = 0; j < n; j++)
			{
				char c;
				cin>>c;
				if (c == 'Y') a[i][j] = 0;
				else if (c == 'B') a[i][j] = 2;
				else if (c == 'E') a[i][j] = 1;
				else if (c == 'R' || c == 'S') a[i][j] = -1;
				else if (c == 'T')
				{
					a[i][j] = 1;
					st = i;
					ed = j;
				}
				v[i][j] = 0;
				dis[i][j] = 10000000;
			}
		}

		djk(0,0);
		if (dis[st][ed] == 10000000) cout<<"Case #"<<test_case<<endl<<"-1"<<endl;
		else
		{
			cout<<"Case #"<<test_case<<endl<<dis[st][ed]<<endl;
		}
	}
	return 0;
}
