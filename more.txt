if (ACTION_DISMISS.equals(intent.getAction())
        || ACTION_SNOOZE.equals(intent.getAction())) {

    context.stopService(
            new Intent(context, AlarmRingingService.class)
    );

    if (ACTION_SNOOZE.equals(intent.getAction())) {
        int minutes = intent.getIntExtra("minutes", 5);
        PendingResult result = goAsync();

        AppExecutors.DB.execute(() -> {
            try {
                AlarmRepository.get(context)
                        .snoozeSync(id, minutes);
            } finally {
                result.finish();
            }
        });

    } else if (id > 0) {
        boolean dismissUpcoming =
                intent.getBooleanExtra(
                        EXTRA_DISMISS_UPCOMING,
                        false
                );

        long triggerAtMillis =
                intent.getLongExtra(
                        EXTRA_DISMISS_TRIGGER_AT,
                        0L
                );

        PendingResult result = goAsync();

        AppExecutors.DB.execute(() -> {
            try {
                if (dismissUpcoming) {
                    AlarmRepository.get(context)
                            .dismissUpcomingSync(
                                    id,
                                    triggerAtMillis
                            );
                } else {
                    /*
                     * Dismiss từ RingActivity.
                     * Giữ nguyên hành vi hiện tại.
                     */
                    AlarmRepository.get(context)
                            .finishOneTimeSync(id);
                }
            } finally {
                result.finish();
            }
        });
    }

    return;
}



public void dismissUpcomingSync(
        int id,
        long triggerAtMillis
) {
    Alarm alarm = dao.getById(id);

    if (alarm == null) {
        return;
    }

    /*
     * Hủy lần báo thức hiện đang được lên lịch,
     * bao gồm cả notification upcoming.
     */
    AlarmScheduler.cancel(context, id);
    UpcomingNotificationManager.cancel(context, id);

    /*
     * triggerAtMillis chính là occurrence mà notification
     * đang đại diện. Lưu nó vào skipUntilMillis để scheduler
     * bỏ qua đúng lần này.
     */
    long dismissedTrigger =
            triggerAtMillis > 0
                    ? triggerAtMillis
                    : AlarmScheduler
                            .nextTriggerIgnoringSkip(alarm);

    alarm.skipUntilMillis = dismissedTrigger;

    /*
     * Báo thức một lần và không giữ lại:
     * chỉ chuyển inactive, không lên lịch lại.
     */
    if (!alarm.repeats()
            && !alarm.keepAfterDismiss) {

        alarm.isActive = false;
        dao.update(alarm);
        return;
    }

    /*
     * Báo thức lặp hoặc báo thức một lần có keepAfterDismiss:
     * vẫn giữ active và lên lịch occurrence tiếp theo.
     */
    dao.update(alarm);

    if (alarm.isActive) {
        AlarmScheduler.schedule(context, alarm);
    }
}
