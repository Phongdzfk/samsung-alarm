#include <iostream>
using namespace std;
long long a[105];
long long visited[105];
int kqua[105];
long long n;
long long ans;

struct point{ int x,y; };
point q[9000005];
int front_, rear_;
void push(int x,int y){ rear_++; q[rear_].x=x; q[rear_].y=y; }
point pop(){ return q[front_++]; }
bool empty_(){ return front_ > rear_; }
int check = 0;

void quaylui(long long i, long long total,long long cnt)
{
	if (i == n)
	{
		if (total > ans) ans = total;
		return;
	}
	for (int j = 0; j < cnt; j++)
	{
		long long tmp = total;
		if (j == 0 && cnt - 1 != 0) tmp += a[1];
		else if (j == cnt -1 && cnt - 1 != 0) tmp+= a[cnt-2];
		else if (cnt - 1 == 0)
		{
			tmp += a[j];
		}
		else
		{
			tmp = tmp + (a[j-1] * a[j+1]);
		}
		int spare = a[j];
		for (int z = j; z < cnt - 1; z++)
		{
			a[z] = a[z+1];
		}
		quaylui(i+1,tmp,cnt-1);
		for (int z = j + 1; z < cnt; z++)
		{
			a[z] = a[z-1];
		}
		a[j] = spare;
	}
	return;
}

int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	//freopen("Text.txt","r",stdin);
	int t;
	cin>>t;
	for (int test=1; test<=t; test++)
	{
		check = 1;
		ans = 0;
		cin>>n;
		for (int i = 0; i < n; i++)
		{
			visited[i] = 0;
			cin>>a[i];
		}
		quaylui(0,0,n);
		cout<<"Case #"<<test<<endl<<ans<<endl;
	}
	return 0;
}
