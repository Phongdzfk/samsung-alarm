#include <iostream>
using namespace std;
int a[3005][3005] = {};
int dis[3005][3005] = {};
int visited[3005] = {};
int n,p,q,s,t;
int ny[3005] = {};
int nx[3005] = {};
int rear = -1;
int front = 0;
int ans;
int abss(int a,int b)
{
	return max(a-b,b-a);
}
struct point
{
	int x;
	int y;
};
point qu[1000005];
void push (int x, int y)
{
	rear++;
	qu[rear].x = x;
	qu[rear].y = y;
}
point pop()
{
	return qu[front++];
}
bool empty()
{
	return front > rear;
}
void quaylui(int i,int total,int cnt)
{
	if (cnt == n)
	{
		if (dis[i][n+1] + total < ans) ans = dis[i][n+1] + total;
		return;
	}
	if (total > ans) return;
	for (int j = 1; j <= n; j++)
	{
		if (visited[j] == 0)
		{
			visited[j] = 1;
			quaylui(j,total + dis[i][j],cnt+1);
			visited[j] = 0;
		}
	}
	return;
}
int main()
{
	freopen("Text.txt","r",stdin);
	int t1;
	cin>>t1;
	for (int test = 1; test <= t1; test++)
	{
		ans = 1e9;
		cin>>p>>q>>s>>t>>n;
		nx[0] = p;
		ny[0] = q;
		for (int i = 1; i <= n; i++)
		{
			int x,y;
			cin>>x>>y;
			nx[i] = x;
			ny[i] = y;
		}
		nx[n+1] = s;
		ny[n+1] = t;
		for (int i = 0; i <= n+1; i++)
		{
			visited[i] = 0;
			for (int j = 0; j <= n+1; j++)
			{
				dis[i][j] = 0;
			}
		}
		for (int i = 0; i <= n+1; i++)
		{
			for (int j = i + 1; j <= n+1; j++)
			{
				dis[i][j] = abss(nx[j],nx[i]) + abss(ny[j],ny[i]);
				dis[j][i] = dis[i][j];
			}
		}
		visited[0] = 1;
		quaylui(0,0,0);
		cout<<"Case #"<<test<<" "<<ans<<endl;
	}
	return 0;
}
