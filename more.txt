You are given 28 different types of domino, each domino has size of 1x2 with 2 numbers on it as follow





And a board of size 7x8, your task is cover the board with above dominos such that a domino can only be placed on two adjacent squares on board if the numbers of the squares and of domino are equal.



How many different way to cover the board?



For example if the board are given as below, there are 18 way to cover it, one of them is as below.





Input



The first line is the number of test case T (T < = 50).



Each test case will be given on 7 lines, each line have 8 numbers separate by a space is the board for the test case. It have a empty line between 2 test cases.



Output



Print each test case on one line, the first is "#x", where x is the test case number, separate by a space and the numbers of different way to cover the board.



#include <iostream>
using namespace std;
long long a[10][10];
long long visited[10][10];
long long checked[10][10];
int dx[4] = {0,0,-1,1};
int dy[4] = {-1,1,0,0};

int ans;
long long n,k;

struct point{ int x,y; };
point q[9000005];
int front_, rear_;
void push(int x,int y){ rear_++; q[rear_].x=x; q[rear_].y=y; }
point pop(){ return q[front_++]; }
bool empty_(){ return front_ > rear_; }

void quaylui(int cnt)
{
	if (cnt == 56)
	{
		ans++;
		return;
	}
	for (int i = 0; i < 7; i++)
	{
		for (int j = 0; j < 8; j++)
		{
			if (visited[i][j] == 0)
			{
				int check = 0;
				for (int z = 0; z < 4; z++)
				{
					int xx = i + dx[z];
					int yy = j + dy[z];
					if (xx < 0 || yy < 0 || xx >= 7 || yy >=8) continue;
					if (visited[xx][yy] == 1) continue;
					if (checked[a[i][j]][a[xx][yy]] == 1) continue;
					check = 1;
					visited[i][j] = 1;
					visited[xx][yy] = 1;
					checked[a[i][j]][a[xx][yy]] = 1;
					checked[a[xx][yy]][a[i][j]] = 1;
					quaylui(cnt+2);
					visited[i][j] = 0;
					visited[xx][yy] = 0;
					checked[a[i][j]][a[xx][yy]] = 0;
					checked[a[xx][yy]][a[i][j]] = 0;
				}
				if (check == 0) return;
			}
		}
	}
	return;
}

int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	//freopen("Text.txt","r",stdin);
	int t;
	cin>>t;
	for (int test=1; test<=t; test++)
	{
		ans = 0;
		for (int i = 0; i < 7; i++)
		{
			for (int j = 0; j < 8; j++)
			{
				cin>>a[i][j];
				visited[i][j] = 0;
				checked[i][j] = 0;
			}
		}
		quaylui(0);
		cout<<"#"<<test<<" "<<ans<<endl;
	}
	return 0;
}
