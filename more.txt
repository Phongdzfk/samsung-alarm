#include <iostream>
using namespace std;
long long a[20];
int dx[4] = {0,0,-1,1};
int dy[4] = {-1,1,0,0};
int unique[500][120];

long long visited[20];
int ans;
long long n,sum;

struct point{ int x,y; };
point q[9000005];
int front_, rear_;
void push(int x,int y){ rear_++; q[rear_].x=x; q[rear_].y=y; }
point pop(){ return q[front_++]; }
bool empty_(){ return front_ > rear_; }

void quaylui(int x,int tong)
{
	if (x == n)
	{
		if (tong == sum) 
		{
			int check[105] = {};
			for (int i = 0; i < n; i ++)
			{
				if (visited[i] == 1)
				{
					check[a[i]]++;
				}
			}
			for (int z = 1; z <= ans; z++)
			{
				int kt = 0;
				for (int i = 0; i < 105; i ++)
				{
					if (check[i] != unique[z][i])
					{
						kt = 1;
						break;
					}
				}
				if (kt == 0) return;
			}
			ans++;
			for (int i = 0; i < 105; i ++)
			{
				unique[ans][i] = check[i];
			}
		}
		return;
	}
	if (tong > sum) return;
	for (int i = 0; i <=1; i++)
	{
		int z = visited[x];
		visited[x] = i;
		quaylui(x+1,tong + a[x] * i);
		visited[x] = z;
	}
}

int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	int t;
	cin>>t;
	for (int test=1; test<=t; test++)
	{
		ans = 0;
		cin>>sum>>n;
		for (int i = 0; i < n; i++)
		{
			cin>>a[i];
		}
		for (int i = 0; i < 5000; i++)
		{
			for (int j = 0; j < 105; j++)
			{
				unique[i][j] = 0;
			}
		}
		quaylui(0,0);
		if (ans == 0) cout<<"#"<<test<<" "<<-1<<endl;
		else cout<<"#"<<test<<" "<<ans<<endl;
	}
	return 0;
}
