int khachlen(int x, int pri)
{
	int cnt = 0;
	int acc = 1;
	int sokhach = cua[x];
	int vitri = vitricua[x];
	if (chongoi[vitri] == 0) 
	{
		chongoi[vitri] = 1;
		sokhach -= 1;
		cnt+=acc;
	}
	int l = vitri - 1;
	int r = vitri + 1;
	while(sokhach > 1)
	{
		acc++;
		if (l >=1 && chongoi[l] == 0)
		{
			chongoi[l] = 1;
			cnt+=acc;
			sokhach--;
		}
		if (r <= n && chongoi[r] == 0)
		{
			chongoi[r] = 1;
			cnt+=acc;
			sokhach--;
		}
		l--;
		r++;
		if (l < 1 && r > n) break;
	}
	if (sokhach > 0)
	{
		acc++;
		int accl;
		int accr;
		accl = accr = acc;
		while (r <= n && chongoi[r] == 1) 
		{
			r++;
			accr++;
		}
		while (l >= 1 && chongoi[l] == 1) 
		{
			l--;
			accl++;
		}
		if (l < 1) 
		{
			chongoi[r] = 1;
			cnt+=accr;
		}
		else if ( r> n)
		{
			chongoi[l] = 1;
			cnt+=accl;
		}
		else if (l >= 1 && r <=n)
		{
			if (accl > accr)
			{
				chongoi[r] = 1;
				cnt+=accr;
			}
			else if (accl < accr)
			{
				chongoi[l] = 1;
				cnt+=accl;
			}
			else if (accl == accr)
			{
				if (pri == 0)
				{
					chongoi[l] = 1;
					cnt+=accl;
				}
				else if (pri == 1)
				{
					chongoi[r] = 1;
					cnt+=accr;
				}
			}
		}
	}
	return cnt;
}
void quaylui(int pri,int total,int dem)
{
	if (total > ans) return;
	if (dem == 3)
	{
		if (total < ans) ans = total;
	}
	for (int i = 1; i <=3; i++)
	{
		if (visited[i] == 0)
		{
			visited[i] = 1;
			int saved[100] = {};
			for (int x = 1; x <=n; x++)
			{
				saved[x] = chongoi[x];
			}
			int tmp = khachlen(i,pri);
			quaylui(0,total + tmp,dem+1);
			quaylui(1,total + tmp,dem+1);
			visited[i] = 0;
			for (int x = 1; x <= n; x++)
			{
				chongoi[x] = saved[x];
			}
		}
	}
	return;
}

#include <iostream>

int n,e,k,t;

float a[105][105];
int front_ = 0;
int rear_ = -1;
float xacsuat[105][1105];
bool visited[105][1105];

struct point
{
	int node;
	int time;
};
point q[300005];
void push(int x,int tm)
{
	rear_++;
	q[rear_].node = x;
	q[rear_].time = tm;
}
point pop()
{
	return q[front_++];
}
bool empty_()
{
	return front_ > rear_;
}

void bfs(int start,int startTime)
{
	front_ = 0; rear_ = -1;
	xacsuat[start][startTime] = 1.0;
	push(start,startTime);
	visited[start][startTime] = true;

	while(!empty_())
	{
		point z = pop();
		int sr = z.node;
		int tm = z.time;
		if (tm > t) continue;

		for (int i = 1; i <= n; i++)
		{
			if (a[sr][i] != -1)
			{
				int nt = tm + 10;
				xacsuat[i][nt] += xacsuat[sr][tm] * a[sr][i];
				if (!visited[i][nt])
				{
					visited[i][nt] = true;
					push(i,nt);
				}
			}
		}
	}
}
