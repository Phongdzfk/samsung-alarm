#include <iostream>
using namespace std;
int a[3005][3005] = {};
int visited[3005][3005] = {};
int n;
int dx[4] = {0,0,1,-1};
int dy[4] = {1,-1,0,0};
int rear = -1;
int front = 0;
int ans;
int abss(int a,int b)
{
	return max(a-b,b-a);
}
struct point
{
	int x;
	int y;
};
point qu[1000005];
void push (int x, int y)
{
	rear++;
	qu[rear].x = x;
	qu[rear].y = y;
}
point pop()
{
	return qu[front++];
}
bool empty()
{
	return front > rear;
}
void quaylui(int x,int y, int hmax,int hmin)
{
	if (x == n-1 && y == n-1)
	{
		if (abss(hmax,hmin) < ans) ans = abss(hmax,hmin);
		return;
	}
	if (abss(hmax,hmin) > ans) return;
	int check[200][200] = {};
	int checkmin = 1e9;
	for (int i = 0; i < 4; i++)
	{
		int x1 = x + dx[i];
		int x2 = y + dy[i];
		if (x1 <0 || x2 < 0|| x1 >= n || x2 >= n) continue;
		if (visited[x1][x2] == 1) continue;
		if (a[x1][x2] >= hmin  && a[x1][x2] <= hmax)
		{
			check[x1][x2] = 0;
		}
		if (a[x1][x2] > hmax)
		{
			check[x1][x2] = a[x1][x2] - hmax;
		}
		if (a[x1][x2] < hmin)
		{
			check[x1][x2] = hmin - a[x1][x2];
		}
		if (check[x1][x2] < checkmin) checkmin = check[x1][x2];
	}
	for (int i = 0; i < 4; i++)
	{
		int x1 = x + dx[i];
		int x2 = y + dy[i];
		if (x1 <0 || x2 < 0|| x1 >= n || x2 >= n) continue;
		if (visited[x1][x2] == 1) continue;
		if (check[x1][x2] == checkmin)
		{
			visited[x1][x2] = 1;
			quaylui(x1,x2,max(hmax,a[x1][x2]),min(hmin,a[x1][x2]));
		}
		if (check[x1][x2] != checkmin && x1 == n-1 && x2 == n-1)
		{
			visited[x1][x2] = 1;
			quaylui(x1,x2,max(hmax,a[x1][x2]),min(hmin,a[x1][x2]));
		}
	}
	return;
}
int main()
{
	//freopen("Text.txt","r",stdin);
	int t1;
	cin>>t1;
	for (int test = 1; test <= t1; test++)
	{
		cin>>n;
		ans = 1e9;
		for (int i = 0; i < n; i++)
		{
			for (int j = 0; j < n; j++)
			{
				cin>>a[i][j];
				visited[i][j] = 0;
			}
		}
		quaylui(0,0,a[0][0],a[0][0]);
		cout<<"#"<<test<<" "<<ans<<endl;
	}
	return 0;
}
