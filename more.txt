int best = 1e9;
int board[4][4]; // trạng thái gốc

int dx[] = {0, 0, 0, -1, 1};
int dy[] = {0, -1, 1, 0, 0};

void press(int r, int c) {
    for (int k = 0; k < 5; k++) {
        int nr = r + dx[k], nc = c + dy[k];
        if (nr >= 0 && nr < 4 && nc >= 0 && nc < 4)
            board[nr][nc] ^= 1;
    }
}

bool isSameColor() {
    for (int i = 0; i < 4; i++)
        for (int j = 0; j < 4; j++)
            if (board[i][j] != board[0][0]) return false;
    return true;
}

void backtrack(int idx, int count) {
    if (count >= best) return; // prune nhẹ nếu đã tệ hơn best
    if (idx == 16) {
        if (isSameColor()) best = min(best, count);
        return;
    }
    int r = idx / 4, c = idx % 4;

    // Không bấm ô này
    backtrack(idx + 1, count);

    // Bấm ô này
    press(r, c);
    backtrack(idx + 1, count + 1);
    press(r, c); // undo (bấm lại lần nữa = trả về trạng thái cũ)
}
