#include <iostream>
using namespace std;
long long a[30][30];
int visited[30][30];
long long n,m;

int quaylui(int x, int cnt)
{
	int best_ = cnt;
	for (int i = 0; i < n; i++)
	{
		if (a[x][i] == 1 && visited[x][i] == 0)
		{
			visited[x][i] = 1;
			visited[i][x] = 1;
			int tmp = quaylui(i, cnt+1);
			if (tmp > best_) best_ = tmp;
			visited[x][i] = 0;
			visited[i][x] = 0;
		}
	}
	return best_;
}

int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	int t;
	cin>>t;
	for (int test=1; test<=t; test++)
	{
		int ans = 0;
		cin>>n>>m;
		for (int i = 0; i < n; i++)
			for (int j = 0; j < n; j++)
			{
				a[i][j] = 0;
				visited[i][j] = 0;
			}
		for (int i = 0; i <m; i++)
		{
			int x,y;
			cin>>x>>y;
			a[x][y] = 1;
			a[y][x] = 1;
		}
		for (int i = 0; i < n; i++)
		{
			int res = quaylui(i,0);
			if (res > ans) ans = res;
		}
		cout<<ans<<endl;
	}
	return 0;
}
