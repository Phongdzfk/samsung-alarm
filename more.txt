#include <iostream>
using namespace std;
int n,e,k,t;
double ans1;
int ans2;
double a[105][105];
int front = 0;
int rear = -1;
int visited[105];
double xacsuat[105][1025];
struct point
{
	int a;
	int b;
};
int q[1000005];
void push(int x)
{
	q[++rear] = x;
}
int pop()
{
	return q[front++];
}
bool empty()
{
	return front > rear;
}
void bfs()
{
	while(!empty())
	{
		int sr = pop();
		if (t >= visited[sr] - 10 && t < visited[sr])
		{
			if (ans1 < xacsuat[sr][t])
			{
				ans1 = xacsuat[sr][t];
				ans2 = sr;
			}
			continue;
		}
		if (t < visited[sr] - 10) continue;
		for (int i = 1; i <= n; i++)
		{
			if (a[sr][i] != -1)
			{
				visited[i] = visited[sr] + 10;
				push(i);
				if (xacsuat[i][visited[i]-1] == 0)
				{
					double tmp = xacsuat[sr][visited[i] - 11] * a[sr][i];
					for (int z = visited[i] - 10; z < visited[i]; z++)
					{
						xacsuat[i][z] = tmp;
					}
				}
				else if (xacsuat[i][visited[i]-1] != 0)
				{
					double tmp = xacsuat[i][visited[i]-1] + (xacsuat[sr][visited[i]-11] * a[sr][i]);
					for (int z = visited[i] - 10; z < visited[i]; z++)
					{
						xacsuat[i][z] = tmp;
					}
				}
			}
		}
	}
}
int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	for (int test=1; test<=10; test++)
	{
		cin>>n>>e>>k>>t;
		cin>>n;
		front = 0;
		rear = -1;
		for (int i = 1; i <=n ;i++)
		{
			loop[i] = -1;
			visited[i] = -1;
			for (int j = 1; j <= n; j++)
			{
				a[i][j] = -1;
			}
		}
		for (int i = 1; i <= n; i++)
		{
			for (int j = 0; j <= t + 15; j++)
			{
				xacsuat[i][j] = 0;
			}
		}
		for (int i = 1; i <= e; i++)
		{
			int source, des, value;
			cin>>source>>des>>value;
			a[source][des] = value;
		}
		push(1);
		visited[1] = 10;
		for (int i = 0; i < 10; i++)
		{
			xacsuat[1][i] = 1.0;
		}
		ans1 = 0.0;
		ans2 = -1;
		bfs();
		cout.precision(6);
		cout<<"#"<<test<<" "<<ans2 - 1<<" ";
		cout<<fixed<<ans1<<" ";
		front = 0;
		rear = -1;
		for (int i = 1; i <=n ;i++)
		{
			visited[i] = -1;
		}
		for (int i = 1; i <= n; i++)
		{
			for (int j = 0; j <= t + 15; j++)
			{
				xacsuat[i][j] = 0;
			}
		}
		push(1);
		visited[1] = 10 + k;
		for (int i = k; i < 10 + k; i++)
		{
			xacsuat[1][i] = 1.0;
		}
		ans1 = -1;
		ans2 = -1;
		bfs();
		cout.precision(6);
		cout<<ans2 -1<<" ";
		cout<<fixed<<ans1<<endl;
	}
	return 0;
}
