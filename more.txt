Cho một số tự nhiên N ≤ 9. Giữa các số từ 1 đến N hãy thêm vào các dấu + và - sao cho kết quả thu được bằng 0. Hãy viết chương trình tìm tất cả các khả năng có thể.

[Input]

Dòng đầu tiên là T số testcase. T dòng tiếp theo là các số tự nhiên N <= 9.

 

[Output]

Mỗi test case in ra “# ” theo sau là số lượng kết quả tìm được mỗi test case. 



[Sample] 

[Input]

1

7

 

[Output]

#1 6

 

Giải thích

1-2-3-4-5+6+7=0

1-2+3+4-5+6-7=0

1-23-45+67=0

1-23+4+5+6+7=0

1+2-3-4+5+6-7=0

1+2-3+4-5-6+7=0

void quaylui(int idx, int total,int acc,int check)
{
	if (idx == n + 1)
	{
		if (total == 0) ans++;
		return;
	}
	for (int i = 0; i <= 2; i++)
	{
		if (i == 0)
		{
			int tmp = total;
			if (acc == idx) 
			{
				tmp += acc;
			}
			else if (acc != idx)
			{
				if (check == 1)
				{
					tmp = tmp - acc + idx;
				}
				else tmp = tmp + acc + idx;
			}
			quaylui(idx+1,tmp,idx+1,i);
		}
		else if (i == 1)
		{
			int tmp = total;
			if (acc == idx) 
			{
				tmp += acc;
			}
			else if (acc != idx)
			{
				if (check == 1)
				{
					tmp = tmp - acc - idx;
				}
				else tmp = tmp + acc - idx;
			}
			quaylui(idx+1,tmp,idx+1,i);
		}
		else if (i == 2)
		{
			if (acc == idx)
			{
				quaylui(idx+1,total,(acc-1) * 10 + acc,check);
			}
			else
			{
				quaylui(idx+1,total,(acc * 10) + idx,check);
			}
		}
	}
}
