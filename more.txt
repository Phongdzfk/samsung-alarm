#include <iostream>
using namespace std;
long long m,d;
int mana[6];
int tt[6];
int x[25];
int dx[8] = {0,0,1,-1,1,1,-1,-1};
int dy[8] = {1,-1,0,0,1,-1,1,-1};
int ans;
int maxcnt;
void quaylui(int loai,int energy,int thoigian,int km)
{
	if (energy < 0) return;
	if (loai == 5)
	{
		if (thoigian < ans) ans = thoigian;
		return;
	}
	for (int i = km; i <= d; i++)
	{
		quaylui(loai+1,energy-(mana[loai] * (i-km)),thoigian+(tt[loai] * (i-km)),i + 1);
	}
}

int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	//freopen("Text.txt","r",stdin);
	int t;
	cin>>t;
	for (int test=1; test<=t; test++)
	{
		ans = 1e9;
		cin>>m>>d;
		for (int i = 1; i <= 5; i++)
		{
			int p,s;
			cin>>p>>s;
			int tmp = p*60 + s;
			tt[i] = tmp;
			cin>>mana[i];
		}
		quaylui(1,m,0,1);
		if (ans == 1e9) cout<<"Case #"<<test<<endl<<-1<<endl;
		else
		{
			int phut = ans/60;
			int giay = ans%60;
			cout<<"Case #"<<test<<endl<<phut<<" "<<giay<<endl;
		}
	}
	return 0;
}
