#include <iostream>
using namespace std;
long long n;
int ans;

// idx: chữ số bắt đầu toán hạng tiếp theo (1..n)
// total: tổng biểu thức tính tới hiện tại
void quaylui(int idx, long long total)
{
	if (idx == n + 1)
	{
		if (total == 0) ans++;
		return;
	}

	// dựng 1 toán hạng bằng cách ghép các chữ số idx, idx+1, ..., j
	long long operand = 0;
	for (int j = idx; j <= n; j++)
	{
		operand = operand * 10 + j;   // ghép thêm chữ số j vào cuối toán hạng

		if (idx == 1)
		{
			// chữ số đầu tiên: không có dấu đứng trước -> luôn cộng
			quaylui(j+1, total + operand);
		}
		else
		{
			quaylui(j+1, total + operand);   // đặt dấu +
			quaylui(j+1, total - operand);   // đặt dấu -
		}
	}
}

int main()
{
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	int t;
	cin>>t;
	for (int test=1; test<=t; test++)
	{
		ans = 0;
		cin>>n;
		quaylui(1, 0);
		cout<<"#"<<test<<" "<<ans<<endl;
	}
	return 0;
}
