#include <iostream>
using namespace std;
int a[100005] = {};
int k;
int maxx;
int n;
int ans = 0;
int sum;
int res = 0;
void swap(int a, int b)
{
	int tmp = a;
	a = b;
	b = tmp;
}
int check(int x)
{
	int cnt = 0;
	int summ = 0;
	for (int i = 0; i < n; i++)
	{
		if (summ+a[i] > x)
		{
			summ = 0;
			cnt++;
		}
		else
		{
			summ+=a[i];
		}
	}
	return cnt;
}
void binas(int mi, int ma)
{
	if (mi == ma) return;
	int mid = (mi+ma)/2;
	int tmp = check(mid);
	if (tmp == k)
	{
		ans = max(mid,ans);
		binas(mi,mid-1);
	}
	else if (tmp < k) binas(mi,mid-1);
	else binas(mid+1,ma);
	return;
}
int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
	cin>>t;
	for (int test = 1; test <= t; test++)
	{
		maxx = 0;
		ans = 0;
		sum = 0;
		cin>>k>>n;
		for (int i = 0; i < n; i++)
		{
			cin>>a[i];
			if (a[i] > maxx) maxx = a[i];
			sum += a[i];
		}
		binas(maxx,sum);
		cout<<ans<<endl;
		
	}
    return 0;
}
